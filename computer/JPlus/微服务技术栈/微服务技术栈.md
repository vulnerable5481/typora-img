# 三.MQ



## 1.同步通信的缺点

同步通讯的缺点:1.**耗时长！**比如支付服务需要调用其他服务模块的方法也就是调用Feign,**耗时会叠加**

​							2.**耦合度高！**每新增一个和支付相关的业务逻辑，都需要修改支付服务的代码

​							3.**资源浪费** 调用链中的每个服务在等待响应中，不能释放请求占用的资源，高并发下极度浪费
​							4.**级联失败**  如果调用链中某一个服务失败，所有的调用方都会跟着失败,迅速导致整个微服务群故障

## 3.进阶检索(DSL)



### 3.1 QueryDSL

<font color='red'>**注意match里面只能写一个检索条件！**</font>

```
http://192.168.200.130:9200/bank/_search
```

#### ①match_all

```
一般情况都是将查询条件写在请求体，
{
	"query":{
		//全部检索
		"match_all": {} //没有条件就是查询所有
	},
	"sort": [
		{
			"account_name": "asc"  //根据account_name 升序排列
		},
		{
			"balance": "desc"     //根据 balance 降序排列
		}
	],
	"_sorce":["balance","firstName"]     //类似于select balance,firstName from xxx,规定查询数据的那些属性
	"from":5,	//分页，从下标为5的数据开始显示
	"size":10   //显示十条数据
}
```

#### ②match


```
{
	"query":{
	//全文检索 会按照评分来排序,会对检索条件进行分词匹配, 这个评分就是根据倒排索引来计算出来的相关性！
	"match":{
		"address":"mill lane" //查询地址包含mill/lane  的文档
	}
	}
}
```

#### ③match_phrase


```
{
	"query":{
	//短语检索
	"match_phrase":{  
		"address":"mill lane" //如果使用短语检索就会将mill lane当成一个完整的词来匹配
	}
	}
}
```

#### ④multi_match


```
{
	"query":{
		//多字段检索同一值
		"multi_match":{
			"query":"mill lane",//注意此处是开启分词检索
			"fields":["address","city"]  //只要address city 其中之一满足检索条件即可
		}
	}
}
```

#### ⑤bool复合检索

```
{
	"query":{
		"bool":{
			"must":[  //必须满足的条件
				{
					"range":{  // 18<=age<=30
						"age":{
							"gte":18,
							"lte":30  
						}
					},
					"match":{
						"address":"mill"
					}
				}
			],
			"must_not":[ //必须不满足的条件
				"match":"{
					"gender":"Male"
				}"
			],
			"should":[//应该满足的条件，满足可以加分，不满足也没有影响
				"match":{
					"firstName":"z"
				}
			]
		}
	}
}
```



#### ⑥filter过滤

```
{
    "query": {
        "bool": {
            "filter": {     //如果使用filter即使满足条件也不会加分的；must,should等则都会加分
                "range": {    
                    "age": {          
                        "gte": 18,    
                        "lte": 30     
                    }
                }
            }
        }
    }
}

```



#### ⑦term

<font color='orange'>**全文检索用match, 其他非text字段使用term 精确匹配**</font>

```
{
	"query":{
		"term":{  
			"account_number":20   //像这种非字符的就推荐使用term
		},
		"match":{
			"address":"mile lane"  //这种可以分词查询的字符text之类的 就推荐使用match
		}
	}
}

```
